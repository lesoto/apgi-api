35def configure_distributed_tracing(): 

36 """ 

37 Configure OpenTelemetry distributed tracing for the application. 

38 

39 Sets up tracing with Jaeger and OTLP exporters based on environment configuration. 

40 """ 

41 if not OPENTELEMETRY_AVAILABLE: 

42 print("OpenTelemetry not available, skipping tracing configuration") 

43 return 

44 

45 # Get tracing configuration from environment 

46 tracing_enabled = os.getenv("TRACING_ENABLED", "false").lower() == "true" 

47 jaeger_endpoint = os.getenv("JAEGER_ENDPOINT", "http://localhost:14268/api/traces") 

48 otlp_endpoint = os.getenv("OTLP_ENDPOINT", "http://localhost:4317") 

49 service_name = os.getenv("TRACING_SERVICE_NAME", "apgi-api") 

50 service_version = os.getenv("API_VERSION", "1.0.0") 

51 

52 if not tracing_enabled: 

53 return 

54 

55 # Set up tracer provider 

56 trace.set_tracer_provider( 

57 TracerProvider( 

58 resource=Resource.create( 

59 { 

60 "service.name": service_name, 

61 "service.version": service_version, 

62 } 

63 ) 

64 ) 

65 ) 

66 

67 # Configure Jaeger exporter 

68 jaeger_exporter = JaegerExporter( 

69 agent_host_name="localhost", 

70 agent_port=14268, 

71 ) 

72 

73 # Configure OTLP exporter 

74 otlp_exporter = OTLPSpanExporter( 

75 endpoint=otlp_endpoint, 

76 insecure=True, 

77 ) 

78 

79 # Add span processors 

80 span_processor_jaeger = BatchSpanProcessor(jaeger_exporter) 

81 span_processor_otlp = BatchSpanProcessor(otlp_exporter) 

82 

83 tracer_provider = typing.cast(TracerProvider, trace.get_tracer_provider()) 

84 tracer_provider.add_span_processor(span_processor_jaeger) 

85 tracer_provider.add_span_processor(span_processor_otlp)

91 

92 This should be called after the FastAPI app is created but before it starts serving requests. 

93 """ 

94 if not OPENTELEMETRY_AVAILABLE: 

95 print("OpenTelemetry not available, skipping application instrumentation")

98 tracing_enabled = os.getenv("TRACING_ENABLED", "false").lower() == "true" 

99 

100 if not tracing_enabled: 

101 return

103 # Instrument FastAPI
