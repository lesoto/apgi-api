33def configure_distributed_tracing( 

34 service_name: str = "apgi-api", 

35 service_version: str = "1.0.0", 

36 jaeger_endpoint: Optional[str] = None, 

37 otlp_endpoint: Optional[str] = None, 

38 enable_console_exporter: bool = False, 

39 sampling_rate: float = 1.0, 

40): 

41 """ 

42 Configure distributed tracing with OpenTelemetry. 

43 

44 Args: 

45 service_name: Name of the service 

46 service_version: Version of the service 

47 jaeger_endpoint: Jaeger collector endpoint (optional) 

48 otlp_endpoint: OTLP gRPC endpoint (optional) 

49 enable_console_exporter: Whether to export traces to console 

50 sampling_rate: Sampling rate for traces (0.0 to 1.0) 

51 """ 

52 if not OPENTELEMETRY_AVAILABLE: 

53 logger.warning("OpenTelemetry not available, skipping tracing configuration") 

54 return 

55 

56 try: 

57 # Set up resource 

58 resource = Resource.create( 

59 { 

60 SERVICE_NAME: service_name, 

61 SERVICE_VERSION: service_version, 

62 } 

63 ) 

64 

65 # Create tracer provider 

66 trace.set_tracer_provider(TracerProvider(resource=resource)) 

67 

68 # Add span processors based on configuration 

69 tracer_provider = trace.get_tracer_provider() 

70 

71 # Console exporter for development 

72 if enable_console_exporter or False: 

73 console_processor = BatchSpanProcessor(ConsoleSpanExporter()) 

74 tracer_provider.add_span_processor(console_processor) # type: ignore[attr-defined] 

75 logger.info("Console trace exporter enabled") 

76 

77 # Jaeger exporter 

78 if jaeger_endpoint: 

79 jaeger_exporter = JaegerExporter( 

80 agent_host_name=( 

81 jaeger_endpoint.split(":")[0] if ":" in jaeger_endpoint else jaeger_endpoint 

82 ), 

83 agent_port=int(jaeger_endpoint.split(":")[1]) if ":" in jaeger_endpoint else 6831, 

84 ) 

85 jaeger_processor = BatchSpanProcessor(jaeger_exporter) 

86 tracer_provider.add_span_processor(jaeger_processor) # type: ignore[attr-defined] 

87 logger.info(f"Jaeger trace exporter configured at {jaeger_endpoint}") 

88 

89 # OTLP gRPC exporter 

90 if otlp_endpoint: 

91 otlp_exporter = OTLPSpanExporter( 

92 endpoint=otlp_endpoint, 

93 insecure=True, # For development; use TLS in production 

94 ) 

95 otlp_processor = BatchSpanProcessor(otlp_exporter) 

96 tracer_provider.add_span_processor(otlp_processor) # type: ignore[attr-defined] 

97 logger.info(f"OTLP gRPC trace exporter configured at {otlp_endpoint}") 

98 

99 # Instrument frameworks 

100 # FastAPI instrumentation 

101 FastAPIInstrumentor.instrument_app( 

102 server_request_hook=_server_request_hook,

119 )

126def _server_request_hook(span, scope): 

127 """Hook to add request-specific information to spans.""" 

128 if span and span.is_recording(): 

129 # Add route information 

130 if hasattr(scope, "get") and callable(scope.get): 

131 route = scope.get("route") 

132 if route: 

133 span.set_attribute( 

134 "http.route", route.path if hasattr(route, "path") else str(route) 

135 ) 

136 

137 # Add user information if available 

138 if hasattr(scope, "get") and callable(scope.get): 

139 user = scope.get("user") 

140 if user: 

141 span.set_attribute("user.id", getattr(user, "id", None)) 

142 span.set_attribute("user.username", getattr(user, "username", None)) 

143

145def _client_request_hook(span, method, url): 

146 """Hook for client requests.""" 

147 if span and span.is_recording(): 

148 span.set_attribute("http.method", method) 

149 span.set_attribute("http.url", str(url))

152def _client_response_hook(span, response_headers, response_body): 

153 """Hook for client responses.""" 

154 if span and span.is_recording(): 

155 span.set_attribute("http.response_complete", True)
